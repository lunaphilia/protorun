### Fixes applied (all fn keywords removed from code blocks):
- let println = fn(self, ...) -> Unit  →  let println = (self, ...) -> Unit
- let map = fn[T, U](...)              →  let map = [T, U](...)
- let getOrElse = fn[T](...)          →  let getOrElse = [T](...)
- let main = fn() -> Unit = {...}     →  let main = () -> Unit = {...}  (4 occurrences)
- map(maybeNumber, fn(x) = ...)       →  map(maybeNumber, (x) = ...)
- numbers.map(fn(x) = ...)            →  numbers.map((x) = ...)
- numbers.filter(fn(x) = ...)         →  numbers.filter((x) = ...)
